# MCP 四层一致 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让四层 `mcp.json` 真正生效：解析合并、凭据引用、stdio 连接、工具注册、按会话过滤、生命周期管理，并用测试与手册锁定边界。

**Architecture:** 仓库内实现最小 MCP stdio 客户端（JSON-RPC 2.0），启动时把四层配置读成快照并并行连接；每个 server 生成一个 Pi Extension，MCP 扩展不进入默认扩展集，会话创建时按「系统 → 空间 → 员工 → 项目」合并结果选择。MCP 管理器负责健康检查、退避重连与退出清理；任何 server 故障都只影响自己。

**Tech Stack:** Node.js ≥ 22.19、TypeScript（strip-only 运行）、Vitest、TypeBox、`@earendil-works/pi-durable` 扩展机制、Node `child_process`。

**Spec:** `docs/superpowers/specs/2026-10-17-mcp-design.md`

## Global Constraints

- Node.js ≥ 22.19；继续使用 `node src/server/main.ts` 的 strip-only 运行方式，不引入需要编译的语法（不要用参数属性、不要用装饰器）。
- 不新增任何 npm 依赖；MCP 协议由仓库内代码实现。
- 只支持 stdio transport；不实现 HTTP / SSE / OAuth / resources / prompts / sampling。
- `mcp.json` 不允许明文凭据；凭据只从 `.openstaffs/config.json` 的 `mcp.<server>.<key>` 读取，占位符为 `${config.mcp.<server>.<key>}`。
- 同名 server 合并顺序：系统 → 空间 → 员工 → 项目，后者胜。
- 所有新增面向用户的日志与工具错误使用中文；错误文本不得包含凭据值。
- 进程启动使用 `shell: false`；POSIX 下 detached 启动并按进程组清理。
- MCP 扩展不进入默认扩展集；工具清单与扩展选择随启动 / 新会话固化。
- MCP 工具总数 > 20 时全部 `deferred: true`，并安装 `tool_search`。
- 每个任务结束前必须让对应测试与 `npm run typecheck` 通过，再提交。

## Review Focus

- **同名 server 跨层覆盖**：项目层定义必须压过员工、空间、系统；私聊不能看到项目层——由 Task 1 的合并测试锁定。
- **凭据出现在 `args` / `env` / `cwd`**：任何字符串字段都要插值，缺失时必须报错而不是留空或回落——由 Task 1 的凭据测试锁定。
- **MCP server 卡死 / 命令不存在**：启动不能被无限拖住，失败后会话仍可创建——由 Task 2 与 Task 4 的超时测试锁定。
- **员工自挂 server 的越界**：员工目录里的 `mcp.json` 只能进入该员工自己的新会话，不能泄漏给其他员工——由 Task 4 的隔离测试锁定。
- **工具清单分页与重名**：分页必须全部取回；同一 server 内重名只能保留一个，不能产生重复工具注册——由 Task 2 与 Task 3 锁定。

```ts
/** MCP 凭据：config.json 的 mcp.<server>.<key>（机器私有；独立读取，未配置模式也可用）。 */
export type McpCredentials = Record<string, Record<string, string>>;

export function readMcpCredentials(ws: Workspace): McpCredentials {
  let raw = "";
  try {
    raw = fs.readFileSync(configFile(ws), "utf8");
  } catch {
    return {};
  }
  try {
    const parsed = JSON.parse(raw) as { mcp?: unknown };
    if (typeof parsed.mcp !== "object" || parsed.mcp === null || Array.isArray(parsed.mcp)) return {};
    const out: McpCredentials = {};
    for (const [rawServer, entry] of Object.entries(parsed.mcp)) {
      const server = rawServer.trim();
      if (server === "" || typeof entry !== "object" || entry === null || Array.isArray(entry)) continue;
      const values: Record<string, string> = {};
      for (const [rawKey, value] of Object.entries(entry)) {
        const key = rawKey.trim();
        const text = requiredString(value);
        if (key !== "" && text !== undefined) values[key] = text;
      }
      if (Object.keys(values).length > 0) out[server] = values;
    }
    return out;
  } catch {
    return {};
  }
}
```

```ts
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { readMcpCredentials, type McpCredentials } from "./config.ts";
import { listEmployees } from "./employee.ts";
import { employeeDir, projectDir, spaceDir, type Workspace } from "./paths.ts";
import { listProjects } from "./project.ts";
import type { Relation } from "./types.ts";

export const SYSTEM_MCP_FILE = fileURLToPath(new URL("../mcp.json", import.meta.url));
export const DEFAULT_MCP_REQUEST_TIMEOUT_MS = 60_000;
export const MCP_DEFERRED_THRESHOLD = 20;

export type McpServerConfig = {
  name: string;
  command: string;
  args: string[];
  env: Record<string, string>;
  cwd: string;
  timeoutMs: number;
};

export type McpScope =
  | { kind: "system" }
  | { kind: "space" }
  | { kind: "employee"; id: string }
  | { kind: "project"; id: string };

export type McpSource = {
  scope: McpScope;
  file: string;
  servers: McpServerConfig[];
  invalid: string[];
  errors: string[];
};

export type McpSnapshot = { sources: McpSource[] };

export type McpSelection = { name: string; scope: McpScope; config: McpServerConfig };

export type McpSessionResolver = {
  extensionsFor(relation: Relation, speakerId: string): string[];
};

export function mcpScopeKey(scope: McpScope): string {
  switch (scope.kind) {
    case "system":
      return "system";
    case "space":
      return "space";
    case "employee":
      return `employee:${scope.id}`;
    case "project":
      return `project:${scope.id}`;
  }
}

export function mcpExtensionName(scope: McpScope, server: string): string {
  return `mcp:${mcpScopeKey(scope)}:${server}`;
}

function safeSegment(value: string): string {
  const cleaned = value.replace(/[^A-Za-z0-9_-]/gu, "_");
  return cleaned === "" ? "unnamed" : cleaned;
}

export function mcpToolName(server: string, tool: string): string {
  return `mcp__${safeSegment(server)}__${safeSegment(tool)}`;
}

function isObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function requiredString(value: unknown): string | undefined {
  return typeof value === "string" && value.trim() !== "" ? value.trim() : undefined;
}

const CREDENTIAL_REF = /\$\{config\.mcp\.([A-Za-z0-9_-]+)\.([^}]+)\}/gu;

function interpolate(
  value: string,
  credentials: McpCredentials,
  errors: string[],
): string | undefined {
  let missing: string | undefined;
  const out = value.replace(CREDENTIAL_REF, (_whole, server: string, key: string) => {
    const resolved = credentials[server]?.[key];
    if (resolved === undefined || resolved === "") {
      missing ??= `${server}.${key}`;
      return "";
    }
    return resolved;
  });
  if (missing !== undefined) {
    errors.push(`缺少凭据：\${config.mcp.${missing}}（请在 .openstaffs/config.json 的 mcp.${missing} 配置）`);
    return undefined;
  }
  return out;
}

function parseServer(
  name: string,
  file: string,
  value: unknown,
  credentials: McpCredentials,
): { config?: McpServerConfig; error?: string } {
  if (!isObject(value)) return { error: "配置必须是对象" };
  const interpolateErrors: string[] = [];
  const commandRaw = requiredString(value.command);
  if (commandRaw === undefined) return { error: "缺少 command" };
  const command = interpolate(commandRaw, credentials, interpolateErrors);
  if (command === undefined) return { error: interpolateErrors.join("；") };

  const args: string[] = [];
  if (value.args !== undefined) {
    if (!Array.isArray(value.args)) return { error: "args 必须是字符串数组" };
    for (const arg of value.args) {
      if (typeof arg !== "string") return { error: "args 必须是字符串数组" };
      const resolved = interpolate(arg, credentials, interpolateErrors);
      if (resolved === undefined) return { error: interpolateErrors.join("；") };
      args.push(resolved);
    }
  }

  const env: Record<string, string> = {};
  if (value.env !== undefined) {
    if (!isObject(value.env)) return { error: "env 必须是对象" };
    for (const [rawKey, rawValue] of Object.entries(value.env)) {
      if (typeof rawValue !== "string") return { error: `env.${rawKey} 必须是字符串` };
      const resolved = interpolate(rawValue, credentials, interpolateErrors);
      if (resolved === undefined) return { error: interpolateErrors.join("；") };
      env[rawKey] = resolved;
    }
  }

  let cwd = path.dirname(file);
  if (value.cwd !== undefined) {
    if (typeof value.cwd !== "string" || value.cwd.trim() === "") return { error: "cwd 必须是非空字符串" };
    const resolved = interpolate(value.cwd.trim(), credentials, interpolateErrors);
    if (resolved === undefined) return { error: interpolateErrors.join("；") };
    cwd = path.resolve(path.dirname(file), resolved);
  }

  let timeoutMs = DEFAULT_MCP_REQUEST_TIMEOUT_MS;
  if (value.timeoutMs !== undefined) {
    if (typeof value.timeoutMs !== "number" || !Number.isInteger(value.timeoutMs) || value.timeoutMs <= 0) {
      return { error: "timeoutMs 必须是正整数" };
    }
    timeoutMs = value.timeoutMs;
  }

  return { config: { name, command, args, env, cwd, timeoutMs } };
}

export function readMcpSource(scope: McpScope, file: string, credentials: McpCredentials): McpSource {
  const source: McpSource = { scope, file, servers: [], invalid: [], errors: [] };
  let raw = "";
  try {
    raw = fs.readFileSync(file, "utf8");
  } catch {
    return source;
  }
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    source.errors.push(`MCP 配置不是合法 JSON：${file}`);
    return source;
  }
  if (!isObject(parsed) || parsed.mcpServers === undefined) return source;
  if (!isObject(parsed.mcpServers)) {
    source.errors.push(`MCP 配置的 mcpServers 必须是对象：${file}`);
    return source;
  }
  for (const [rawName, value] of Object.entries(parsed.mcpServers)) {
    const name = rawName.trim();
    if (name === "") {
      source.invalid.push(rawName);
      source.errors.push(`MCP server 名字不能为空：${file}`);
      continue;
    }
    const parsedServer = parseServer(name, file, value, credentials);
    if (parsedServer.config === undefined) {
      source.invalid.push(name);
      source.errors.push(`MCP server "${name}" 配置错误（${file}）：${parsedServer.error ?? "未知错误"}`);
      continue;
    }
    source.servers.push(parsedServer.config);
  }
  return source;
}

export function loadMcpSnapshot(ws: Workspace, options?: { systemFile?: string }): McpSnapshot {
  const credentials = readMcpCredentials(ws);
  const sources: McpSource[] = [];
  sources.push(readMcpSource({ kind: "system" }, options?.systemFile ?? SYSTEM_MCP_FILE, credentials));
  sources.push(readMcpSource({ kind: "space" }, path.join(spaceDir(ws), "mcp.json"), credentials));
  for (const employee of listEmployees(ws)) {
    sources.push(
      readMcpSource({ kind: "employee", id: employee.id }, path.join(employeeDir(ws, employee.id), "mcp.json"), credentials),
    );
  }
  for (const project of listProjects(ws)) {
    sources.push(
      readMcpSource({ kind: "project", id: project.id }, path.join(projectDir(ws, project.id), "mcp.json"), credentials),
    );
  }
  return { sources };
}

function sourcesFor(snapshot: McpSnapshot, relation: Relation, speakerId: string): McpSource[] {
  const out = snapshot.sources.filter((source) => source.scope.kind === "system" || source.scope.kind === "space");
  for (const source of snapshot.sources) {
    if (source.scope.kind === "employee" && source.scope.id === speakerId) out.push(source);
  }
  if (relation.kind === "group") {
    for (const source of snapshot.sources) {
      if (source.scope.kind === "project" && source.scope.id === relation.project) out.push(source);
    }
  }
  return out;
}

export function resolveMcpServers(
  snapshot: McpSnapshot,
  relation: Relation,
  speakerId: string,
): { selections: McpSelection[] } {
  const merged = new Map<string, McpSelection | "invalid">();
  for (const source of sourcesFor(snapshot, relation, speakerId)) {
    for (const name of source.invalid) merged.set(name, "invalid");
    for (const config of source.servers) {
      merged.set(config.name, { name: config.name, scope: source.scope, config });
    }
  }
  return {
    selections: [...merged.values()].filter((value): value is McpSelection => value !== "invalid"),
  };
}
```

```js
}

function handleCall(message) {
  const name = message.params?.name;
  const args = message.params?.arguments ?? {};
  if (name === "echo") {
    if (outputFile !== undefined) fs.writeFileSync(outputFile, `echo:${args.text ?? ""}\n`);
    send({ jsonrpc: "2.0", id: message.id, result: { content: [{ type: "text", text: `echo:${args.text ?? ""}` }] } });
    return;
  }
  if (name === "fail") {
    send({ jsonrpc: "2.0", id: message.id, result: { content: [{ type: "text", text: "boom" }], isError: true } });
    return;
  }
  if (name === "image") {
    send({
      jsonrpc: "2.0",
      id: message.id,
      result: { content: [{ type: "image", data: "aGk=", mimeType: "image/png" }] },
    });
    return;
  }
  if (name === "slow") {
    setTimeout(() => {
      send({ jsonrpc: "2.0", id: message.id, result: { content: [{ type: "text", text: "slow-done" }] } });
    }, 500);
    return;
  }
  send({
    jsonrpc: "2.0",
    id: message.id,
    result: { content: [{ type: "text", text: `unknown:${name}` }], isError: true },
  });
}

function handle(message) {
  if (message.method === "initialize") {
    send({
      jsonrpc: "2.0",
      id: message.id,
      result: {
        protocolVersion: "2024-11-05",
        capabilities: { tools: {} },
        serverInfo: { name: "fixture", version: "1.0.0" },
      },
    });
    return;
  }
  if (message.method === "notifications/initialized") return;
  if (message.method === "ping") {
    send({ jsonrpc: "2.0", id: message.id, result: {} });
    return;
  }
  if (message.method === "tools/list") {
    if (mode === "paged") {
      const start = Number(message.params?.cursor ?? 0);
      const page = tools().slice(start, start + 2);
      send({
        jsonrpc: "2.0",
        id: message.id,
        result: { tools: page, ...(start + 2 < tools().length ? { nextCursor: String(start + 2) } : {}) },
      });
      return;
    }
    send({ jsonrpc: "2.0", id: message.id, result: { tools: tools() } });
    return;
  }
  if (message.method === "tools/call") {
    handleCall(message);
    return;
  }
  if (message.id !== undefined) {
    send({ jsonrpc: "2.0", id: message.id, error: { code: -32601, message: "method not found" } });
  }
}

if (mode === "hang") {
  setInterval(() => {}, 1000);
} else {
  if (mode === "crash") {
```

```ts
import { spawn, type ChildProcessWithoutNullStreams } from "node:child_process";

export type McpToolInfo = {
  name: string;
  description?: string;
  inputSchema?: unknown;
};

export type McpCallResult = {
  content: readonly unknown[];
  structuredContent?: unknown;
  isError?: boolean;
};

export type McpClientOptions = {
  serverName: string;
  command: string;
  args: readonly string[];
  env: NodeJS.ProcessEnv;
  cwd: string;
  requestTimeoutMs?: number;
  connectTimeoutMs?: number;
  onExit?: (reason: string) => void;
  log?: (line: string) => void;
};

type Pending = {
  resolve: (value: unknown) => void;
  reject: (error: Error) => void;
  timer: NodeJS.Timeout;
};

const DEFAULT_REQUEST_TIMEOUT_MS = 60_000;
const DEFAULT_CONNECT_TIMEOUT_MS = 10_000;

function isObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function signalProcessGroup(child: ChildProcessWithoutNullStreams, signal: NodeJS.Signals): void {
  const pid = child.pid;
  if (pid === undefined) return;
  if (process.platform !== "win32") {
    try {
      process.kill(-pid, signal);
      return;
    } catch {
      // 进程组不存在时回落到单进程
    }
  }
  try {
    child.kill(signal);
  } catch {
    // 已退出
  }
}

export class McpStdioClient {
  readonly #options: McpClientOptions;
  #child: ChildProcessWithoutNullStreams | undefined;
  #buffer = "";
  #nextId = 1;
  #pending = new Map<number, Pending>();
  #ready = false;
  #closed = false;
  #exitNotified = false;

  constructor(options: McpClientOptions) {
    this.#options = options;
  }

  #log(line: string): void {
    (this.#options.log ?? ((text) => console.error(text)))(line);
  }

  async start(): Promise<void> {
    if (this.#child !== undefined) throw new Error(`MCP server "${this.#options.serverName}" 已经启动`);
    const child = spawn(this.#options.command, [...this.#options.args], {
      cwd: this.#options.cwd,
      env: this.#options.env,
      stdio: ["pipe", "pipe", "pipe"],
      shell: false,
      detached: process.platform !== "win32",
      windowsHide: true,
    });
    this.#child = child;
    child.stdout.setEncoding("utf8");
    child.stdout.on("data", (chunk: string) => this.#onStdout(chunk));
    child.stderr.setEncoding("utf8");
    child.stderr.on("data", (chunk: string) => {
      for (const line of chunk.split(/\r?\n/u)) {
        if (line.trim() !== "") this.#log(`[mcp:${this.#options.serverName}] ${line.slice(0, 2000)}`);
      }
    });
    child.on("error", (error) => {
      this.#failAll(new Error(`MCP server "${this.#options.serverName}" 启动失败：${error.message}`));
    });
    child.on("exit", (code, signal) => {
      const reason = `进程退出（code=${code ?? "null"}, signal=${signal ?? "null"}）`;
      this.#ready = false;
      this.#child = undefined;
      this.#failAll(new Error(`MCP server "${this.#options.serverName}" ${reason}`));
      if (!this.#closed && !this.#exitNotified) {
        this.#exitNotified = true;
        this.#options.onExit?.(reason);
      }
    });

    try {
      await this.#request(
        "initialize",
        {
          protocolVersion: "2024-11-05",
          capabilities: {},
          clientInfo: { name: "openstaffs", version: "0.0.0" },
        },
        this.#options.connectTimeoutMs ?? DEFAULT_CONNECT_TIMEOUT_MS,
        "初始化超时",
      );
      this.#notify("notifications/initialized", {});
      this.#ready = true;
    } catch (error) {
      await this.close();
      throw error;
    }
  }

  #assertReady(): void {
    if (!this.#ready || this.#child === undefined) {
      throw new Error(`MCP server "${this.#options.serverName}" 未启动或未就绪`);
    }
  }

  #write(message: unknown): void {
    const child = this.#child;
    if (child === undefined || this.#closed) {
      throw new Error(`MCP server "${this.#options.serverName}" 未启动`);
    }
    child.stdin.write(`${JSON.stringify(message)}\n`);
  }

  #notify(method: string, params: Record<string, unknown>): void {
    this.#write({ jsonrpc: "2.0", method, params });
  }

  #request(
    method: string,
    params: Record<string, unknown>,
    timeoutMs: number,
    timeoutLabel = "请求超时",
  ): Promise<unknown> {
    const id = this.#nextId++;
    return new Promise<unknown>((resolve, reject) => {
      const timer = setTimeout(() => {
        if (!this.#pending.has(id)) return;
        this.#pending.delete(id);
        reject(new Error(`MCP server "${this.#options.serverName}" ${timeoutLabel}：${method}（${timeoutMs}ms）`));
      }, timeoutMs);
      this.#pending.set(id, { resolve, reject, timer });
      try {
        this.#write({ jsonrpc: "2.0", id, method, params });
      } catch (error) {
        clearTimeout(timer);
        this.#pending.delete(id);
        reject(error instanceof Error ? error : new Error(String(error)));
      }
    });
  }

  #onStdout(chunk: string): void {
    this.#buffer += chunk;
    for (;;) {
      const index = this.#buffer.indexOf("\n");
      if (index < 0) break;
      const line = this.#buffer.slice(0, index).trim();
      this.#buffer = this.#buffer.slice(index + 1);
      if (line === "") continue;
      let message: unknown;
      try {
        message = JSON.parse(line);
      } catch {
        this.#log(`[mcp:${this.#options.serverName}] 忽略非 JSON 输出：${line.slice(0, 200)}`);
        continue;
      }
      this.#onMessage(message);
    }
  }

  #onMessage(message: unknown): void {
    if (!isObject(message)) return;
    if (typeof message.method === "string") {
      if (message.id !== undefined) {
        try {
          this.#write({
            jsonrpc: "2.0",
            id: message.id,
            error: { code: -32601, message: `不支持 server 发起的请求：${message.method}` },
          });
        } catch {
          // 连接已断
        }
      }
      return;
    }
    if (typeof message.id !== "number") return;
    const pending = this.#pending.get(message.id);
    if (pending === undefined) return;
    this.#pending.delete(message.id);
    clearTimeout(pending.timer);
    if (isObject(message.error)) {
      const text = typeof message.error.message === "string" ? message.error.message : "未知错误";
      pending.reject(new Error(`MCP server "${this.#options.serverName}" 返回错误：${text}`));
      return;
    }
    pending.resolve(message.result);
  }

  #failAll(error: Error): void {
    for (const pending of this.#pending.values()) {
      clearTimeout(pending.timer);
      pending.reject(error);
    }
    this.#pending.clear();
  }

  async listTools(): Promise<McpToolInfo[]> {
    this.#assertReady();
    const tools: McpToolInfo[] = [];
    const seenCursors = new Set<string>();
    let cursor: string | undefined;
    for (;;) {
      const result = await this.#request(
        "tools/list",
        cursor === undefined ? {} : { cursor },
        this.#options.requestTimeoutMs ?? DEFAULT_REQUEST_TIMEOUT_MS,
      );
      if (!isObject(result) || !Array.isArray(result.tools)) {
        throw new Error(`MCP server "${this.#options.serverName}" tools/list 返回格式不合法`);
      }
      for (const item of result.tools) {
        if (!isObject(item) || typeof item.name !== "string" || item.name.trim() === "") continue;
        tools.push({
          name: item.name,
          ...(typeof item.description === "string" ? { description: item.description } : {}),
          ...(item.inputSchema === undefined ? {} : { inputSchema: item.inputSchema }),
        });
      }
      const next = typeof result.nextCursor === "string" && result.nextCursor !== "" ? result.nextCursor : undefined;
      if (next === undefined || seenCursors.has(next)) break;
      seenCursors.add(next);
      cursor = next;
    }
    return tools;
  }

  async callTool(name: string, args: Record<string, unknown>): Promise<McpCallResult> {
    this.#assertReady();
    const result = await this.#request(
      "tools/call",
      { name, arguments: args },
      this.#options.requestTimeoutMs ?? DEFAULT_REQUEST_TIMEOUT_MS,
    );
    if (!isObject(result) || !Array.isArray(result.content)) {
      throw new Error(`MCP server "${this.#options.serverName}" tools/call 返回格式不合法`);
    }
    return {
      content: result.content,
      ...(result.structuredContent === undefined ? {} : { structuredContent: result.structuredContent }),
      ...(result.isError === true ? { isError: true } : {}),
    };
  }

  async ping(): Promise<void> {
    this.#assertReady();
    await this.#request("ping", {}, 5000);
  }

  async close(): Promise<void> {
    this.#closed = true;
    this.#ready = false;
    const child = this.#child;
    this.#child = undefined;
    this.#failAll(new Error(`MCP server "${this.#options.serverName}" 已关闭`));
    if (child === undefined || child.exitCode !== null || child.signalCode !== null) return;
    await new Promise<void>((resolve) => {
      let done = false;
      const finish = () => {
        if (done) return;
        done = true;
        resolve();
      };
      child.once("exit", finish);
      signalProcessGroup(child, "SIGTERM");
      const timer = setTimeout(() => {
        signalProcessGroup(child, "SIGKILL");
        finish();
      }, 2000);
      timer.unref();
    });
  }
}
```

```ts
import { defineExtension, defineTool, type Extension } from "@earendil-works/pi-durable";
import { Type, type TSchema } from "typebox";
import { mcpToolName } from "../domain/mcp.ts";
import type { McpCallResult, McpToolInfo } from "./mcp-client.ts";

export type McpExtensionOptions = {
  scopeKey: string;
  serverName: string;
  tools: readonly McpToolInfo[];
  deferred: boolean;
  call: (tool: string, args: Record<string, unknown>) => Promise<McpCallResult>;
  log?: (line: string) => void;
};

export type McpToolOutput =
  | { type: "text"; text: string }
  | { type: "image"; data: string; mimeType: string };

function isObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function parametersOf(schema: unknown): TSchema {
  if (isObject(schema) && schema.type === "object") return schema as unknown as TSchema;
  return Type.Object({});
}

export function toToolResult(result: McpCallResult): {
  output: McpToolOutput[];
  isError?: boolean;
} {
  const output: McpToolOutput[] = [];
  for (const block of result.content) {
    if (!isObject(block)) {
      output.push({ type: "text", text: String(block) });
      continue;
    }
    if (block.type === "text" && typeof block.text === "string") {
      output.push({ type: "text", text: block.text });
      continue;
    }
    if (block.type === "image" && typeof block.data === "string" && typeof block.mimeType === "string") {
      output.push({ type: "image", data: block.data, mimeType: block.mimeType });
      continue;
    }
    output.push({ type: "text", text: JSON.stringify(block) });
  }
  if (output.length === 0) {
    output.push({
      type: "text",
      text: result.structuredContent === undefined ? "（无输出）" : JSON.stringify(result.structuredContent),
    });
  }
  return { output, ...(result.isError === true ? { isError: true } : {}) };
}

export function createMcpExtension(options: McpExtensionOptions): Extension {
  const log = options.log ?? ((line: string) => console.error(line));
  const seen = new Set<string>();
  const tools = [];
  for (const tool of options.tools) {
    const name = mcpToolName(options.serverName, tool.name);
    if (seen.has(name)) {
      log(`[mcp:${options.serverName}] 工具重名，跳过：${tool.name}`);
      continue;
    }
    seen.add(name);
    tools.push(
      defineTool({
        name,
        description: tool.description?.trim() || `MCP server "${options.serverName}" 的工具 ${tool.name}`,
        parameters: parametersOf(tool.inputSchema),
        ...(options.deferred ? { deferred: true as const } : {}),
        async execute(args) {
          try {
            const result = await options.call(tool.name, args as Record<string, unknown>);
            return toToolResult(result);
          } catch (error) {
            const message = error instanceof Error ? error.message : String(error);
            return {
              output: [{ type: "text" as const, text: `MCP 工具调用失败（${options.serverName}/${tool.name}）：${message}` }],
              isError: true,
            };
          }
        },
      }),
    );
  }
  return defineExtension({ name: `mcp:${options.scopeKey}:${options.serverName}`, tools });
}
```

```ts
import type { Extension } from "@earendil-works/pi-durable";
import {
  DEFAULT_MCP_REQUEST_TIMEOUT_MS, MCP_DEFERRED_THRESHOLD, loadMcpSnapshot, mcpExtensionName, mcpScopeKey,
  resolveMcpServers, type McpScope, type McpSelection, type McpSessionResolver, type McpSnapshot,
} from "../domain/mcp.ts";
import type { Workspace } from "../domain/paths.ts";
import type { Relation } from "../domain/types.ts";
import { createMcpExtension } from "./mcp-extension.ts";
import { McpStdioClient, type McpCallResult, type McpToolInfo } from "./mcp-client.ts";

export type McpManagerOptions = {
  ws: Workspace;
  /** 系统级 mcp.json 路径（测试注入；缺省 src/mcp.json）。 */
  systemFile?: string;
  log?: (line: string) => void;
  connectTimeoutMs?: number;
  heartbeatMs?: number;
  reconnectBaseMs?: number;
  reconnectMaxAttempts?: number;
  deferredThreshold?: number;
};

type Runtime = {
  key: string;
  serverName: string;
  scope: McpScope;
  selection: McpSelection;
  client?: McpStdioClient;
  tools: McpToolInfo[];
  extension?: Extension;
  ready: boolean;
  reconnectAttempts: number;
  reconnectTimer?: NodeJS.Timeout;
  heartbeatTimer?: NodeJS.Timeout;
  connecting?: Promise<void>;
  stopping: boolean;
};

const ENV_ALLOWLIST = [
  "PATH", "HOME", "USER", "LOGNAME", "SHELL", "TMPDIR", "TMP", "TEMP", "LANG", "LC_ALL", "TZ",
];

function childEnv(configured: Record<string, string>): NodeJS.ProcessEnv {
  const env: NodeJS.ProcessEnv = {};
  for (const key of ENV_ALLOWLIST) {
    const value = process.env[key];
    if (value !== undefined) env[key] = value;
  }
  return { ...env, ...configured };
}

export class McpManager implements McpSessionResolver {
  readonly #ws: Workspace;
  readonly #options: McpManagerOptions;
  #snapshot: McpSnapshot = { sources: [] };
  #runtimes: Runtime[] = [];
  #errors: string[] = [];
  #needsToolSearch = false;
  #closed = false;

  constructor(options: McpManagerOptions) {
    this.#ws = options.ws;
    this.#options = options;
  }

  #log(line: string): void {
    (this.#options.log ?? ((text) => console.error(text)))(line);
  }

  async start(): Promise<void> {
    this.#snapshot = loadMcpSnapshot(this.#ws, {
      ...(this.#options.systemFile === undefined ? {} : { systemFile: this.#options.systemFile }),
    });
    for (const source of this.#snapshot.sources) {
      for (const error of source.errors) this.#errors.push(error);
      for (const error of source.errors) this.#log(`[mcp] ${error}`);
    }

    this.#runtimes = [];
    for (const source of this.#snapshot.sources) {
      for (const config of source.servers) {
        this.#runtimes.push({
          key: mcpExtensionName(source.scope, config.name),
          serverName: config.name,
          scope: source.scope,
          selection: { name: config.name, scope: source.scope, config },
          tools: [],
          ready: false,
          reconnectAttempts: 0,
          stopping: false,
        });
      }
    }

    await Promise.all(
      this.#runtimes.map((runtime) =>
        this.#connect(runtime).catch((error: unknown) => {
          const message = error instanceof Error ? error.message : String(error);
          this.#errors.push(message);
          this.#log(`[mcp] ${message}`);
        }),
      ),
    );

    const totalTools = this.#runtimes.reduce((sum, runtime) => sum + runtime.tools.length, 0);
    this.#needsToolSearch = totalTools > (this.#options.deferredThreshold ?? MCP_DEFERRED_THRESHOLD);

    for (const runtime of this.#runtimes) this.#ensureExtension(runtime);
  }

  #ensureExtension(runtime: Runtime): void {
    if (runtime.tools.length === 0 || runtime.extension !== undefined) return;
    runtime.extension = createMcpExtension({
      scopeKey: mcpScopeKey(runtime.scope),
      serverName: runtime.serverName,
      tools: runtime.tools,
      deferred: this.#needsToolSearch,
      call: (tool, args) => this.#call(runtime, tool, args),
      log: (line) => this.#log(line),
    });
  }

  #connect(runtime: Runtime): Promise<void> {
    if (this.#closed || runtime.stopping) return Promise.resolve();
    if (runtime.ready) return Promise.resolve();
    if (runtime.connecting !== undefined) return runtime.connecting;
    runtime.connecting = (async () => {
      runtime.ready = false;
      this.#stopHeartbeat(runtime);
      const config = runtime.selection.config;
      const client = new McpStdioClient({
        serverName: runtime.serverName,
        command: config.command,
        args: config.args,
        env: childEnv(config.env),
        cwd: config.cwd,
        requestTimeoutMs: config.timeoutMs ?? DEFAULT_MCP_REQUEST_TIMEOUT_MS,
        ...(this.#options.connectTimeoutMs === undefined ? {} : { connectTimeoutMs: this.#options.connectTimeoutMs }),
        log: (line) => this.#log(line),
        onExit: (reason) => {
          if (runtime.stopping || this.#closed) return;
          const wasReady = runtime.ready;
          runtime.ready = false;
          runtime.client = undefined;
          this.#stopHeartbeat(runtime);
          if (!wasReady) return; // 首次连接就没成功：不具备工具清单，重连也无法注册，等重启
          this.#log(`[mcp] server "${runtime.serverName}" ${reason}；准备重连`);
          this.#scheduleReconnect(runtime);
        },
      });
      try {
        await client.start();
        runtime.tools = await client.listTools();
        runtime.client = client;
        runtime.ready = true;
        runtime.reconnectAttempts = 0;
        this.#startHeartbeat(runtime);
      } catch (error) {
        await client.close().catch(() => undefined);
        runtime.ready = false;
        throw error;
      } finally {
        runtime.connecting = undefined;
      }
    })();
    return runtime.connecting;
  }

  #startHeartbeat(runtime: Runtime): void {
    this.#stopHeartbeat(runtime);
    const interval = this.#options.heartbeatMs ?? 30_000;
    if (interval <= 0) return;
    const timer = setInterval(() => {
      void (async () => {
        if (!runtime.ready || runtime.client === undefined) return;
        try {
          await runtime.client.ping();
        } catch (error) {
          const message = error instanceof Error ? error.message : String(error);
          this.#log(`[mcp] server "${runtime.serverName}" 心跳失败：${message}`);
          this.#stopHeartbeat(runtime);
          runtime.ready = false;
          const client = runtime.client;
          runtime.client = undefined;
          await client?.close().catch(() => undefined);
          this.#scheduleReconnect(runtime);
        }
      })();
    }, interval);
    timer.unref();
    runtime.heartbeatTimer = timer;
  }

  #stopHeartbeat(runtime: Runtime): void {
    if (runtime.heartbeatTimer !== undefined) {
      clearInterval(runtime.heartbeatTimer);
      runtime.heartbeatTimer = undefined;
    }
  }

  #scheduleReconnect(runtime: Runtime): void {
    if (this.#closed || runtime.stopping || runtime.reconnectTimer !== undefined) return;
    runtime.reconnectAttempts += 1;
    const max = this.#options.reconnectMaxAttempts ?? 5;
    if (runtime.reconnectAttempts > max) {
      const message = `MCP server "${runtime.serverName}" 重连次数已用尽（${max} 次），需要重启服务`;
      this.#errors.push(message);
      this.#log(`[mcp] ${message}`);
      return;
    }
    const base = this.#options.reconnectBaseMs ?? 1000;
    const delay = Math.min(base * 2 ** (runtime.reconnectAttempts - 1), 30_000);
    const timer = setTimeout(() => {
      runtime.reconnectTimer = undefined;
      this.#connect(runtime).catch((error: unknown) => {
        const message = error instanceof Error ? error.message : String(error);
        this.#errors.push(message);
        this.#log(`[mcp] ${message}`);
        this.#scheduleReconnect(runtime);
      });
    }, delay);
    timer.unref();
    runtime.reconnectTimer = timer;
  }

  async #call(runtime: Runtime, tool: string, args: Record<string, unknown>): Promise<McpCallResult> {
    if (!runtime.ready) {
      try {
        await this.#connect(runtime);
      } catch (error) {
        const message = error instanceof Error ? error.message : String(error);
        throw new Error(`MCP server "${runtime.serverName}" 未连接：${message}`);
      }
    }
    if (!runtime.ready || runtime.client === undefined) {
      throw new Error(`MCP server "${runtime.serverName}" 未连接`);
    }
    return runtime.client.callTool(tool, args);
  }

  extensions(): Extension[] {
    return this.#runtimes.flatMap((runtime) => (runtime.extension === undefined ? [] : [runtime.extension]));
  }

  extensionsFor(relation: Relation, speakerId: string): string[] {
    const byKey = new Map(this.#runtimes.map((runtime) => [runtime.key, runtime]));
    const names: string[] = [];
    for (const selection of resolveMcpServers(this.#snapshot, relation, speakerId).selections) {
      const runtime = byKey.get(mcpExtensionName(selection.scope, selection.name));
      if (runtime?.extension !== undefined) names.push(runtime.extension.name);
    }
    return names;
  }

  needsToolSearch(): boolean {
    return this.#needsToolSearch;
  }

  errors(): string[] {
    return [...this.#errors];
  }

  async close(): Promise<void> {
    this.#closed = true;
    for (const runtime of this.#runtimes) {
      runtime.stopping = true;
      this.#stopHeartbeat(runtime);
      if (runtime.reconnectTimer !== undefined) {
        clearTimeout(runtime.reconnectTimer);
        runtime.reconnectTimer = undefined;
      }
    }
    await Promise.all(
      this.#runtimes.map(async (runtime) => {
        const client = runtime.client;
        runtime.client = undefined;
        runtime.ready = false;
        await client?.close().catch(() => undefined);
      }),
    );
  }
}
```

```ts
async function ensure(
  ws: Workspace, harness: HarnessPort, relation: Relation, speakerId: string,
  defaults?: SessionDefaults, mcp?: McpSessionResolver,
): Promise<string> {
  const speaker = loadEmployee(ws, speakerId);
```
